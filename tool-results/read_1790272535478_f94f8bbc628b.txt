     2→     2→// hollow/solid candles, OHLC bars, grid, period separators, bid line with
     3→     3→// axis tag, crosshair, price/time scales and live tick building.

    10→    10→  id: string;
    11→    11→  type: 'ma';
    12→    12→  label: string;

    15→    15→  color: string;

    18→    18→/** horizontal trade level drawn over the plot (MT5 trade lines):
    19→    19→ *  position entries, S/L, T/P and pending-order prices */

    21→    21→  id: string;
    22→    22→  kind: 'entry' | 'sl' | 'tp' | 'order';

    25→    25→  price: number;

    30→    30→
    31→    31→const PAD_RIGHT = 66; // price axis width
    32→    32→const PAD_BOTTOM = 23; // time axis height
    33→    33→const FONT = '11px Tahoma, "Segoe UI", sans-serif';
    34→    34→
    35→    35→function niceStep(raw: number): number {
    36→    36→  const exp = Math.floor(Math.log10(raw));
    37→    37→  const base = 10 ** exp;
    38→    38→  const frac = raw / base;
    39→    39→  let n: number;
    40→    40→  if (frac <= 1) n = 1;
    41→    41→  else if (frac <= 2) n = 2;
    42→    42→  else if (frac <= 2.5) n = 2.5;
    43→    43→  else if (frac <= 5) n = 5;
    44→    44→  else n = 10;
    45→    45→  return n * base;
    46→    46→}

    49→    49→  60, 120, 180, 300, 600, 900, 1800, 3600, 7200, 10800, 14400, 21600, 43200,
    50→    50→  86400, 172800, 345600, 604800, 1209600, 2592000, 7776000, 15552000, 31536000,
    51→    51→];

    98→    98→  constructor(canvas: HTMLCanvasElement, scheme: ChartScheme) {
    99→    99→    this.canvas = canvas;
   100→   100→    this.scheme = scheme;
   101→   101→    const ctx = canvas.getContext('2d');
   102→   102→    if (!ctx) throw new Error('2d context unavailable');
   103→   103→    this.ctx = ctx;
   104→   104→  }
   105→   105→
   106→   106→  // -------------------------------------------------------------- lifecycle
   107→   107→  resize(w: number, h: number): void {
   108→   108→    this.dpr = window.devicePixelRatio || 1;
   109→   109→    this.width = w;
   110→   110→    this.height = h;
   111→   111→    this.canvas.width = Math.floor(w * this.dpr);
   112→   112→    this.canvas.height = Math.floor(h * this.dpr);
   113→   113→    this.canvas.style.width = `${w}px`;
   114→   114→    this.canvas.style.height = `${h}px`;
   115→   115→    this.invalidate();
   116→   116→  }
   117→   117→
   118→   118→  invalidate(): void {
   119→   119→    this.dirty = true;
   120→   120→    if (this.rafId === null) {
   121→   121→      this.rafId = requestAnimationFrame(() => {
   122→   122→        this.rafId = null;
   123→   123→        if (this.dirty) {
   124→   124→          this.dirty = false;
   125→   125→          this.render();
   126→   126→        }
   127→   127→      });
   128→   128→    }
   129→   129→  }
   130→   130→
   131→   131→  /** enable/disable the candle-close countdown; a private 1s interval
   132→   132→   *  repaints the tag with zero React involvement (no re-render, no lag) */
   133→   133→  setCountdown(on: boolean): void {
   134→   134→    if (on === this.showCountdown) return;
   135→   135→    this.showCountdown = on;
   136→   136→    if (on) {
   137→   137→      this.countdownTimer = setInterval(() => {
   138→   138→        // skip repaints while the page is hidden (battery friendly)
   139→   139→        if (typeof document !== 'undefined' && document.hidden) return;
   140→   140→        this.invalidate();
   141→   141→      }, 1000);
   142→   142→      this.invalidate();
   143→   143→    } else if (this.countdownTimer !== null) {
   144→   144→      clearInterval(this.countdownTimer);
   145→   145→      this.countdownTimer = null;
   146→   146→      this.invalidate();
   147→   147→    }
   148→   148→  }
   149→   149→
   150→   150→  setData(candles: Candle[]): void {
   151→   151→    this.candles = candles;
   152→   152→    this.rightOffset = 0;
   153→   153→    this.autoScroll = true;
   154→   154→    this.invalidate();
   155→   155→  }
   156→   156→
   157→   157→  setQuote(bid: number, ask: number): void {
   158→   158→    this.bid = bid;
   159→   159→    this.ask = ask;
   160→   160→    if (this.candles.length > 0) {
   161→   161→      const last = this.candles[this.candles.length - 1];
   162→   162→      const bucket = Math.floor(Date.now() / 1000 / this.tfSeconds) * this.tfSeconds;
   163→   163→      if (bucket > last.time) {
   164→   164→        this.candles.push({ time: bucket, open: bid, high: bid, low: bid, close: bid, volume: 1 });
   165→   165→      } else {
   166→   166→        last.close = bid;
   167→   167→        if (bid > last.high) last.high = bid;
   168→   168→        if (bid < last.low) last.low = bid;
   169→   169→        last.volume += 1;
   170→   170→      }
   171→   171→    }
   172→   172→    this.invalidate();
   173→   173→  }
   174→   174→
   175→   175→  // ------------------------------------------------------------- interaction
   176→   176→  onWheel(deltaY: number): void {
   177→   177→    const factor = deltaY > 0 ? 1 / 1.18 : 1.18;
   178→   178→    const next = Math.min(48, Math.max(2.2, this.barWidth * factor));
   179→   179→    this.barWidth = next;
   180→   180→    if (this.autoScroll) this.rightOffset = 0;
   181→   181→    this.invalidate();
   182→   182→    this.onViewChange();
   183→   183→  }
   184→   184→
   185→   185→  /** pointer↔price helpers based on the last rendered scale */
   186→   186→  private yFor(price: number): number {
   187→   187→    const { min, max } = this.scale;
   188→   188→    const plotH = this.height - PAD_BOTTOM;
   189→   189→    const f = (price - min) / (max - min || 1);
   190→   190→    // mirrors render yOf — higher prices towards the top
   191→   191→    return Math.round((1 - f) * (plotH - 10) + 5);
   192→   192→  }
   193→   193→
   194→   194→  private priceAtY(y: number): number {
   195→   195→    const { min, max } = this.scale;
   196→   196→    const plotH = this.height - PAD_BOTTOM;
   197→   197→    return min + (1 - (y - 5) / (plotH - 10)) * (max - min);
   198→   198→  }
   199→   199→
   200→   200→  /** the draggable trade line under the pointer, if any */
   201→   201→  private hitLine(x: number, y: number): EngineTradeLine | null {
   202→   202→    const plotW = this.width - PAD_RIGHT;
   203→   203→    if (x > plotW) return null;
   204→   204→    for (const line of this.tradeLines) {
   205→   205→      if (!line.draggable) continue;
   206→   206→      if (Math.abs(y - this.yFor(line.price)) <= 6) return line;
   207→   207→    }
   208→   208→    return null;
   209→   209→  }
   210→   210→
   211→   211→  onMouseDown(x: number, y: number): void {
   212→   212→    // grabbing a trade line takes precedence over chart panning
   213→   213→    const hit = this.hitLine(x, y);
   214→   214→    if (hit) {
   215→   215→      this.dragLineId = hit.id;
   216→   216→      return;
   217→   217→    }
   218→   218→    this.drag = { startX: x, startOffset: this.rightOffset };
   219→   219→  }
   220→   220→
   221→   221→  onMouseMove(x: number, y: number, dragging: boolean): void {
   222→   222→    // live re-pricing of a grabbed trade line
   223→   223→    if (this.dragLineId) {
   224→   224→      const line = this.tradeLines.find((l) => l.id === this.dragLineId);
   225→   225→      if (line) {
   226→   226→        const plotH = this.height - PAD_BOTTOM;
   227→   227→        const yClamped = Math.min(Math.max(y, 5), plotH - 5);
   228→   228→        line.price = Math.max(this.priceAtY(yClamped), 0);
   229→   229→        this.canvas.style.cursor = 'ns-resize';
   230→   230→        this.invalidate();
   231→   231→      }
   232→   232→      return;
   233→   233→    }
   234→   234→    if (dragging && this.drag) {
   235→   235→      const dx = x - this.drag.startX;
   236→   236→      const slots = dx / this.barWidth;
   237→   237→      const plotW = this.width - PAD_RIGHT;
   238→   238→      const maxOffset = this.candles.length - Math.floor(plotW / this.barWidth) - 1;
   239→   239→      // MT5 grab behaviour: the chart follows the pointer — dragging right
   240→   240→      // pulls older bars into view (content moves right with the finger)
   241→   241→      const next = Math.round(this.drag.startOffset + slots);
   242→   242→      this.rightOffset = Math.min(Math.max(next, -3), maxOffset);
   243→   243→      if (this.rightOffset <= 0) {
   244→   244→        // reached the right edge → resume auto scroll
   245→   245→        this.autoScroll = true;
   246→   246→        this.rightOffset = Math.max(this.rightOffset, 0);
   247→   247→      } else {
   248→   248→        this.autoScroll = false;
   249→   249→      }
   250→   250→      this.invalidate();
   251→   251→      this.onViewChange();
   252→   252→      return;
   253→   253→    }
   254→   254→    const plotW = this.width - PAD_RIGHT;
   255→   255→    if (x < plotW && y < this.height - PAD_BOTTOM) {
   256→   256→      this.crosshair = { x, y };
   257→   257→    } else {
   258→   258→      this.crosshair = null;
   259→   259→    }
   260→   260→    // resize cursor over draggable trade lines
   261→   261→    this.canvas.style.cursor = this.hitLine(x, y) ? 'ns-resize' : '';
   262→   262→    // hover callback
   263→   263→    const idx = this.indexAtX(x);
   264→   264→    this.onHover(idx !== null && this.candles[idx] ? this.candles[idx] : null);
   265→   265→    this.invalidate();
   266→   266→  }
   267→   267→
   268→   268→  onMouseLeave(): void {
   269→   269→    this.crosshair = null;
   270→   270→    this.drag = null;
   271→   271→    // cancel any line drag without committing (line snaps back on re-sync)
   272→   272→    this.dragLineId = null;
   273→   273→    this.canvas.style.cursor = '';
   274→   274→    this.onHover(null);
   275→   275→    this.invalidate();
   276→   276→  }
   277→   277→
   278→   278→  onMouseUp(): void {
   279→   279→    if (this.dragLineId) {
   280→   280→      const line = this.tradeLines.find((l) => l.id === this.dragLineId);
   281→   281→      this.dragLineId = null;
   282→   282→      this.canvas.style.cursor = '';
   283→   283→      if (line) {
   284→   284→        const price = Number(line.price.toFixed(this.digits));
   285→   285→        this.onLineRelease(line.id, price);
   286→   286→      }
   287→   287→      return;
   288→   288→    }
   289→   289→    this.drag = null;
   290→   290→  }
   291→   291→
   292→   292→  scrollToEnd(): void {
   293→   293→    this.rightOffset = 0;
   294→   294→    this.autoScroll = true;
   295→   295→    this.invalidate();
   296→   296→  }
   297→   297→
   298→   298→  indexAtX(x: number): number | null {
   299→   299→    const slot = this.barWidth;
   300→   300→    const plotW = this.width - PAD_RIGHT;
   301→   301→    if (x < 0 || x > plotW) return null;
   302→   302→    const rightIdx = this.candles.length - 1 - this.rightOffset;
   303→   303→    // exact inverse of xOf()
   304→   304→    const idx = Math.round(rightIdx - (plotW - this.shiftPx() - slot / 2 - x) / slot);
   305→   305→    if (idx < 0 || idx >= this.candles.length) return null;
   306→   306→    return idx;
   307→   307→  }
   308→   308→
   309→   309→  /** MT5 "chart shift": empty margin at the right edge so the newest
   310→   310→   *  candle (incl. the live bar) renders fully clear of the price axis */
   311→   311→  private shiftPx(): number {
   312→   312→    const plotW = this.width - PAD_RIGHT;
   313→   313→    return Math.round(Math.min(Math.max(plotW * 0.07, 24), 110));
   314→   314→  }
   315→   315→
   316→   316→  // ------------------------------------------------------------------ math
   317→   317→  private visibleRange(): { left: number; right: number } {
   318→   318→    const plotW = this.width - PAD_RIGHT;
   319→   319→    const right = this.candles.length - 1 - this.rightOffset;
   320→   320→    const left = right - Math.ceil((plotW - this.shiftPx()) / this.barWidth) - 1;
   321→   321→    return { left, right };
   322→   322→  }
   323→   323→
   324→   324→  private indicatorSeries: (number | null)[][] = [];
   325→   325→
   326→   326→  private priceRange(left: number, right: number): { min: number; max: number } {
   327→   327→    let min = Infinity;
   328→   328→    let max = -Infinity;
   329→   329→    for (let i = Math.max(0, left); i <= Math.min(this.candles.length - 1, right); i += 1) {
   330→   330→      const c = this.candles[i];
   331→   331→      if (c.low < min) min = c.low;
   332→   332→      if (c.high > max) max = c.high;
   333→   333→    }
   334→   334→    if (this.bid > 0) {
   335→   335→      min = Math.min(min, this.bid);
   336→   336→      max = Math.max(max, this.bid);
   337→   337→    }
   338→   338→    for (const ind of this.indicatorSeries) {
   339→   339→      for (let i = Math.max(0, left); i <= Math.min(ind.length - 1, right); i += 1) {
   340→   340→        const v = ind[i];
   341→   341→        if (v === null) continue;
   342→   342→        if (v < min) min = v;
   343→   343→        if (v > max) max = v;
   344→   344→      }
   345→   345→    }
   346→   346→    if (!isFinite(min) || !isFinite(max)) return { min: 0, max: 1 };
   347→   347→    const pad = (max - min) * 0.08 || max * 0.002;
   348→   348→    return { min: min - pad, max: max + pad };
   349→   349→  }
   350→   350→
   351→   351→  private computeIndicators(): void {
   352→   352→    this.indicatorSeries = this.indicators.map((ind) => {
   353→   353→      const closes = this.candles.map((c) => c.close);
   354→   354→      const out: (number | null)[] = new Array(closes.length).fill(null);
   355→   355→      if (ind.method === 'sma') {
   356→   356→        let sum = 0;
   357→   357→        for (let i = 0; i < closes.length; i += 1) {
   358→   358→          sum += closes[i];
   359→   359→          if (i >= ind.period) sum -= closes[i - ind.period];
   360→   360→          if (i >= ind.period - 1) out[i] = sum / ind.period;
   361→   361→        }
   362→   362→      } else {
   363→   363→        const k = 2 / (ind.period + 1);
   364→   364→        let prev: number | null = null;
   365→   365→        for (let i = 0; i < closes.length; i += 1) {
   366→   366→          prev = prev === null ? closes[i] : closes[i] * k + prev * (1 - k);
   367→   367→          if (i >= ind.period - 1) out[i] = prev;
   368→   368→        }
   369→   369→      }
   370→   370→      return out;
   371→   371→    });
   372→   372→  }
   373→   373→
   374→   374→  // --------------------------------------------------------------- drawing
   375→   375→  render(): void {
   376→   376→    const ctx = this.ctx;
   377→   377→    const W = this.width;
   378→   378→    const H = this.height;
   379→   379→    const plotW = W - PAD_RIGHT;
   380→   380→    const plotH = H - PAD_BOTTOM;
   381→   381→
   382→   382→    ctx.setTransform(this.dpr, 0, 0, this.dpr, 0, 0);
   383→   383→    ctx.font = FONT;
   384→   384→    ctx.imageSmoothingEnabled = false;
   385→   385→
   386→   386→    this.computeIndicators();
   387→   387→
   388→   388→    // background
   389→   389→    ctx.fillStyle = this.scheme.bg;
   390→   390→    ctx.fillRect(0, 0, W, H);
   391→   391→
   392→   392→    if (this.candles.length === 0) return;
   393→   393→
   394→   394→    const { left, right } = this.visibleRange();
   395→   395→    const { min, max } = this.priceRange(left, right);
   396→   396→    // remember the scale for pointer↔price mapping (trade line dragging)
   397→   397→    this.scale = { min, max };
   398→   398→    const slot = this.barWidth;
   399→   399→    const bodyW = Math.max(1, Math.floor(slot) - 2);
   400→   400→    const halfBody = Math.floor(bodyW / 2);
   401→   401→
   402→   402→    const yOf = (price: number) => {
   403→   403→      const f = (price - min) / (max - min);
   404→   404→      // MT5 orientation: higher prices are drawn towards the top
   405→   405→      return Math.round((1 - f) * (plotH - 10) + 5);
   406→   406→    };
   407→   407→    const xOf = (i: number) => {
   408→   408→      const rightIdx = this.candles.length - 1 - this.rightOffset;
   409→   409→      return Math.round(plotW - this.shiftPx() - (rightIdx - i) * slot - slot / 2);
   410→   410→    };
   411→   411→
   412→   412→    // ---- grid ------------------------------------------------------------
   413→   413→    if (this.showGrid) {
   414→   414→      ctx.strokeStyle = this.scheme.grid;
   415→   415→      ctx.lineWidth = 1;
   416→   416→
   417→   417→      const step = niceStep((max - min) / Math.max(2, Math.floor(plotH / 55)));
   418→   418→      ctx.beginPath();
   419→   419→      for (let p = Math.ceil(min / step) * step; p <= max; p += step) {
   420→   420→        const y = yOf(p) + 0.5;
   421→   421→        ctx.moveTo(0, y);
   422→   422→        ctx.lineTo(plotW, y);
   423→   423→      }
   424→   424→      ctx.stroke();
   425→   425→
   426→   426→      const tStep = this.timeStep();
   427→   427→      const leftT = this.candles[Math.max(0, left)]?.time ?? 0;
   428→   428→      const startT = Math.ceil(leftT / tStep) * tStep;
   429→   429→      ctx.beginPath();
   430→   430→      for (let t = startT; ; t += tStep) {
   431→   431→        let idx = this.candles.findIndex((c) => c.time >= t);
   432→   432→        if (idx < 0 || this.candles[idx].time >= t + tStep) break;
   433→   433→        if (idx < left) idx = left;
   434→   434→        const x = xOf(idx) + 0.5;
   435→   435→        if (x < 0 || x > plotW) continue;
   436→   436→        if (idx > right) break;
   437→   437→        ctx.moveTo(x, 0);
   438→   438→        ctx.lineTo(x, plotH);
   439→   439→      }
   440→   440→      ctx.stroke();
   441→   441→    }
   442→   442→
   443→   443→    // ---- period separators ------------------------------------------------
   444→   444→    if (this.showSeparators) {
   445→   445→      ctx.strokeStyle = this.scheme.separator;
   446→   446→      ctx.setLineDash([4, 4]);
   447→   447→      ctx.lineWidth = 1;
   448→   448→      ctx.beginPath();
   449→   449→      const sepUnit = this.tfSeconds >= 86400 ? 'month' : 'day';
   450→   450→      let prevUnit = -1;
   451→   451→      for (let i = Math.max(0, left); i <= Math.min(this.candles.length - 1, right); i += 1) {
   452→   452→        const d = new Date(this.candles[i].time * 1000);
   453→   453→        const unit = sepUnit === 'day' ? d.getUTCDate() : d.getUTCMonth();
   454→   454→        if (prevUnit !== -1 && unit !== prevUnit) {
   455→   455→          const x = xOf(i) + 0.5;
   456→   456→          if (x > 0 && x < plotW) {
   457→   457→            ctx.moveTo(x, 0);
   458→   458→            ctx.lineTo(x, plotH);
   459→   459→          }
   460→   460→        }
   461→   461→        prevUnit = unit;
   462→   462→      }
   463→   463→      ctx.stroke();
   464→   464→      ctx.setLineDash([]);
   465→   465→    }
   466→   466→
   467→   467→    // ---- price series ------------------------------------------------------
   468→   468→    const iStart = Math.max(0, left);
   469→   469→    const iEnd = Math.min(this.candles.length - 1, right);
   470→   470→
   471→   471→    if (this.chartType === 'candles') {
   472→   472→      for (let i = iStart; i <= iEnd; i += 1) {
   473→   473→        const c = this.candles[i];
   474→   474→        const x = xOf(i);
   475→   475→        if (x < -slot || x > plotW + slot) continue;
   476→   476→        const bull = c.close >= c.open;
   477→   477→        const yO = yOf(c.open);
   478→   478→        const yC = yOf(c.close);
   479→   479→        const yH = yOf(c.high);
   480→   480→        const yL = yOf(c.low);
   481→   481→        const top = Math.min(yO, yC);
   482→   482→        const h = Math.max(Math.abs(yC - yO), 1);
   483→   483→
   484→   484→        ctx.strokeStyle = bull ? this.scheme.bullBorder : this.scheme.bearBody;
   485→   485→        ctx.beginPath();
   486→   486→        ctx.moveTo(x + 0.5, yH);
   487→   487→        ctx.lineTo(x + 0.5, yL);
   488→   488→        ctx.stroke();
   489→   489→
   490→   490→        if (bull) {
   491→   491→          // hollow bullish candle — MT5 classic look
   492→   492→          ctx.fillStyle = this.scheme.bullFill;
   493→   493→          ctx.fillRect(x - halfBody, top, bodyW, h);
   494→   494→          ctx.strokeStyle = this.scheme.bullBorder;
   495→   495→          ctx.strokeRect(x - halfBody + 0.5, top + 0.5, bodyW - 1, Math.max(h - 1, 1));
   496→   496→        } else {
   497→   497→          ctx.fillStyle = this.scheme.bearBody;
   498→   498→          ctx.fillRect(x - halfBody, top, bodyW, h);
   499→   499→        }
   500→   500→      }
   501→   501→    } else if (this.chartType === 'bars') {
   502→   502→      for (let i = iStart; i <= iEnd; i += 1) {
   503→   503→        const c = this.candles[i];
   504→   504→        const x = xOf(i);
   505→   505→        if (x < -slot || x > plotW + slot) continue;
   506→   506→        const up = c.close >= c.open;
   507→   507→        ctx.strokeStyle = up ? this.scheme.barUp : this.scheme.barDown;
   508→   508→        const yO = yOf(c.open);
   509→   509→        const yC = yOf(c.close);
   510→   510→        const tick = Math.max(3, Math.floor(slot / 2) - 1);
   511→   511→        ctx.beginPath();
   512→   512→        ctx.moveTo(x + 0.5, yOf(c.high));
   513→   513→        ctx.lineTo(x + 0.5, yOf(c.low));
   514→   514→        ctx.moveTo(x - tick, yO + 0.5);
   515→   515→        ctx.lineTo(x + 0.5, yO + 0.5);
   516→   516→        ctx.moveTo(x + 0.5, yC + 0.5);
   517→   517→        ctx.lineTo(x + tick, yC + 0.5);
   518→   518→        ctx.stroke();
   519→   519→      }
   520→   520→    } else {
   521→   521→      ctx.strokeStyle = this.scheme.line;
   522→   522→      ctx.lineWidth = 1;
   523→   523→      ctx.beginPath();
   524→   524→      let started = false;
   525→   525→      for (let i = iStart; i <= iEnd; i += 1) {
   526→   526→        const x = xOf(i);
   527→   527→        const y = yOf(this.candles[i].close);
   528→   528→        if (!started) {
   529→   529→          ctx.moveTo(x, y);
   530→   530→          started = true;
   531→   531→        } else ctx.lineTo(x, y);
   532→   532→      }
   533→   533→      ctx.stroke();
   534→   534→      ctx.lineWidth = 1;
   535→   535→    }
   536→   536→
   537→   537→    // ---- indicators ---------------------------------------------------------
   538→   538→    this.indicatorSeries.forEach((series, si) => {
   539→   539→      const ind = this.indicators[si];
   540→   540→      if (!ind) return;
   541→   541→      ctx.strokeStyle = ind.color;
   542→   542→      ctx.lineWidth = 1;
   543→   543→      ctx.beginPath();
   544→   544→      let started = false;
   545→   545→      for (let i = iStart; i <= iEnd; i += 1) {
   546→   546→        const v = series[i];
   547→   547→        if (v === null) {
   548→   548→          started = false;
   549→   549→          continue;
   550→   550→        }
   551→   551→        const x = xOf(i);
   552→   552→        const y = yOf(v);
   553→   553→        if (!started) {
   554→   554→          ctx.moveTo(x, y);
   555→   555→          started = true;
   556→   556→        } else ctx.lineTo(x, y);
   557→   557→      }
   558→   558→      ctx.stroke();
   559→   559→    });
   560→   560→
   561→   561→    // ---- bid line + tag ------------------------------------------------------
   562→   562→    if (this.bid > 0 && this.bid >= min && this.bid <= max) {
   563→   563→      const y = yOf(this.bid) + 0.5;
   564→   564→      ctx.strokeStyle = this.scheme.bidLine;
   565→   565→      ctx.setLineDash([5, 3]);
   566→   566→      ctx.beginPath();
   567→   567→      ctx.moveTo(0, y);
   568→   568→      ctx.lineTo(plotW, y);
   569→   569→      ctx.stroke();
   570→   570→      ctx.setLineDash([]);
   571→   571→
   572→   572→      ctx.fillStyle = this.scheme.bidLine;
   573→   573→      ctx.fillRect(plotW + 1, y - 9, PAD_RIGHT - 2, 18);
   574→   574→      ctx.fillStyle = '#FFFFFF';
   575→   575→      ctx.textAlign = 'center';
   576→   576→      ctx.fillText(this.bid.toFixed(this.digits), plotW + PAD_RIGHT / 2, y + 4);
   577→   577→    }
   578→   578→
   579→   579→    // ---- trade levels (MT5 style: dashed line + left label tag) -------------
   580→   580→    // position entries, S/L, T/P and pending-order prices; draggable
   581→   581→    // lines show an axis tag while being re-priced
   582→   582→    for (const tl of this.tradeLines) {
   583→   583→      if (tl.price < min || tl.price > max) continue;
   584→   584→      const y = this.yFor(tl.price) + 0.5;
   585→   585→      ctx.strokeStyle = tl.color;
   586→   586→      ctx.setLineDash([6, 4]);
   587→   587→      ctx.lineWidth = 1;
   588→   588→      ctx.beginPath();
   589→   589→      ctx.moveTo(0, y);
   590→   590→      ctx.lineTo(plotW, y);
   591→   591→      ctx.stroke();
   592→   592→      ctx.setLineDash([]);
   593→   593→
   594→   594→      const text = `${tl.tag} ${tl.price.toFixed(this.digits)}`;
   595→   595→      ctx.fillStyle = tl.color;
   596→   596→      const w = Math.max(58, text.length * 6 + 10);
   597→   597→      ctx.fillRect(2, y - 8, w, 16);
   598→   598→      ctx.fillStyle = '#FFFFFF';
   599→   599→      ctx.textAlign = 'left';
   600→   600→      ctx.fillText(text, 6, y + 4);
   601→   601→
   602→   602→      // dragged line — mirror the live price on the price axis
   603→   603→      if (tl.id === this.dragLineId) {
   604→   604→        ctx.fillStyle = tl.color;
   605→   605→        ctx.fillRect(plotW + 1, y - 9, PAD_RIGHT - 2, 18);
   606→   606→        ctx.fillStyle = '#FFFFFF';
   607→   607→        ctx.textAlign = 'center';
   608→   608→        ctx.fillText(tl.price.toFixed(this.digits), plotW + PAD_RIGHT / 2, y + 4);
   609→   609→      }
   610→   610→    }
   611→   611→
   612→   612→    // ---- price axis ----------------------------------------------------------
   613→   613→    const pStep = niceStep((max - min) / Math.max(2, Math.floor(plotH / 55)));
   614→   614→    for (let p = Math.ceil(min / pStep) * pStep; p <= max; p += pStep) {
   615→   615→      const y = yOf(p);
   616→   616→      ctx.fillStyle = this.scheme.bg;
   617→   617→      ctx.fillRect(plotW + 1, y - 8, PAD_RIGHT - 2, 16);
   618→   618→      ctx.fillStyle = this.scheme.axisText;
   619→   619→      ctx.textAlign = 'left';
   620→   620→      ctx.fillText(p.toFixed(this.digits), plotW + 6, y + 4);
   621→   621→    }
   622→   622→    ctx.strokeStyle = this.scheme.grid;
   623→   623→    ctx.beginPath();
   624→   624→    ctx.moveTo(plotW + 0.5, 0);
   625→   625→    ctx.lineTo(plotW + 0.5, plotH);
   626→   626→    ctx.moveTo(0, plotH + 0.5);
   627→   627→    ctx.lineTo(plotW, plotH + 0.5);
   628→   628→    ctx.stroke();
   629→   629→
   630→   630→    // ---- candle countdown (MT5 price-scale timer) ----------------------------
   631→   631→    // small tag right under the bid tag counting down to the bar close;
   632→   632→    // drawn after the price axis so axis labels never overpaint it
   633→   633→    if (this.showCountdown && this.bid > 0 && this.bid >= min && this.bid <= max) {
   634→   634→      const last = this.candles[this.candles.length - 1];
   635→   635→      const rem = Math.max(0, last.time + this.tfSeconds - Date.now() / 1000);
   636→   636→      const hh = Math.floor(rem / 3600);
   637→   637→      const mm = Math.floor((rem % 3600) / 60);
   638→   638→      const ss = Math.floor(rem % 60);
   639→   639→      const pad2 = (n: number) => String(n).padStart(2, '0');
   640→   640→      const label = hh > 0 ? `${hh}:${pad2(mm)}:${pad2(ss)}` : `${pad2(mm)}:${pad2(ss)}`;
   641→   641→
   642→   642→      const by = yOf(this.bid);
   643→   643→      // bid tag spans by-9 .. by+9 → place the countdown directly below,
   644→   644→      // or flip above when it would run off the bottom edge
   645→   645→      let ty = by + 10;
   646→   646→      if (ty + 16 > plotH) ty = by - 26;
   647→   647→      ctx.fillStyle = '#5A5A5A';
   648→   648→      ctx.fillRect(plotW + 1, ty, PAD_RIGHT - 2, 16);
   649→   649→      ctx.fillStyle = '#FFFFFF';
   650→   650→      ctx.textAlign = 'center';
   651→   651→      ctx.fillText(label, plotW + PAD_RIGHT / 2, ty + 12);
   652→   652→    }
   653→   653→
   654→   654→    // ---- time axis ------------------------------------------------------------
   655→   655→    const tStep = this.timeStep();
   656→   656→    const leftT = this.candles[Math.max(0, left)]?.time ?? 0;
   657→   657→    const startT = Math.ceil(leftT / tStep) * tStep;
   658→   658→    ctx.textAlign = 'center';
   659→   659→    const dayBoundary = 86400;
   660→   660→    for (let t = startT; ; t += tStep) {
   661→   661→      const idx = this.candles.findIndex((c) => c.time >= t);
   662→   662→      if (idx < 0) break;
   663→   663→      if (this.candles[idx].time >= t + tStep) break;
   664→   664→      if (idx > iEnd) break;
   665→   665→      const x = xOf(idx);
   666→   666→      if (x > 0 && x < plotW) {
   667→   667→        const isMidnight = this.candles[idx].time % dayBoundary === 0;
   668→   668→        let label: string;
   669→   669→        if (tStep >= 2592000) {
   670→   670→          label = fmtDayLabelYear(this.candles[idx].time);
   671→   671→        } else if (isMidnight) {
   672→   672→          label = fmtDayLabel(this.candles[idx].time);
   673→   673→        } else {
   674→   674→          label = fmtClock(this.candles[idx].time);
   675→   675→        }
   676→   676→        ctx.fillStyle = this.scheme.axisText;
   677→   677→        ctx.fillText(label, x, plotH + 16);
   678→   678→      }
   679→   679→    }
   680→   680→
   681→   681→    // ---- crosshair -------------------------------------------------------------
   682→   682→    if (this.crosshair) {
   683→   683→      const { x, y } = this.crosshair;
   684→   684→      if (x < plotW && y < plotH) {
   685→   685→        ctx.strokeStyle = this.scheme.crosshair;
   686→   686→        ctx.setLineDash([2, 3]);
   687→   687→        ctx.beginPath();
   688→   688→        ctx.moveTo(x + 0.5, 0);
   689→   689→        ctx.lineTo(x + 0.5, plotH);
   690→   690→        ctx.moveTo(0, y + 0.5);
   691→   691→        ctx.lineTo(plotW, y + 0.5);
   692→   692→        ctx.stroke();
   693→   693→        ctx.setLineDash([]);
   694→   694→
   695→   695→        const price = min + (1 - (y - 5) / (plotH - 10)) * (max - min);
   696→   696→        ctx.fillStyle = '#5A5A5A';
   697→   697→        ctx.fillRect(plotW + 1, y - 9, PAD_RIGHT - 2, 18);
   698→   698→        ctx.fillStyle = '#FFFFFF';
   699→   699→        ctx.textAlign = 'center';
   700→   700→        ctx.fillText(price.toFixed(this.digits), plotW + PAD_RIGHT / 2, y + 4);
   701→   701→
   702→   702→        const idx = this.indexAtX(x);
   703→   703→        if (idx !== null && this.candles[idx]) {
   704→   704→          const d = new Date(this.candles[idx].time * 1000);
   705→   705→          const pad = (n: number) => String(n).padStart(2, '0');
   706→   706→          const label =
   707→   707→            this.tfSeconds >= 86400
   708→   708→              ? `${d.getFullYear()}.${pad(d.getMonth() + 1)}.${pad(d.getDate())}`
   709→   709→              : `${pad(d.getHours())}:${pad(d.getMinutes())}`;
   710→   710→          ctx.fillStyle = '#5A5A5A';
   711→   711→          const w = Math.max(54, label.length * 7 + 10);
   712→   712→          ctx.fillRect(x - w / 2, plotH + 1, w, PAD_BOTTOM - 2);
   713→   713→          ctx.fillStyle = '#FFFFFF';
   714→   714→          ctx.fillText(label, x, plotH + 16);
   715→   715→        }
   716→   716→      }
   717→   717→    }
   718→   718→  }
   719→   719→
   720→   720→  private timeStep(): number {
   721→   721→    const targetPx = 80;
   722→   722→    const slots = targetPx / this.barWidth;
   723→   723→    const targetSec = slots * this.tfSeconds;
   724→   724→    for (const s of TIME_STEPS) {
   725→   725→      if (s >= targetSec) return s;
   726→   726→    }
   727→   727→    return 31536000;
   728→   728→  }
   729→   729→}
